// Fixed Interaction Mechanics: Track mouse drag relative to viewport space let isDragging = false; let previousMousePosition = { x: 0, y: 0 }; canvas.addEventListener('mousedown', (e) => { isDragging = true; previousMousePosition = { x: e.clientX, y: e.clientY }; }); window.addEventListener('mousemove', (e) => { if (!isDragging) return; const deltaX = e.clientX - previousMousePosition.x; const deltaY = e.clientY - previousMousePosition.y; // Create a threshold matrix calculation based on the current top/bottom flip // If the cube is upside down on the X-axis, invert the horizontal panning matrix vector const cosX = Math.cos(cubeRotationX); const directionModifier = cosX > 0 ? 1 : -1; // Apply consistent transformation values cubeRotationY += deltaX * 0.01 * directionModifier; cubeRotationX += deltaY * 0.01; previousMousePosition = { x: e.clientX, y: e.clientY }; }); window.addEventListener('mouseup', () => isDragging = false); // Add touch support for mobile browsers too canvas.addEventListener('touchstart', (e) => { if (e.touches.length === 1) { isDragging = true; previousMousePosition = { x: e.touches[0].clientX, y: e.touches[0].clientY }; } }); window.addEventListener('touchmove', (e) => { if (!isDragging || e.touches.length !== 1) return; const deltaX = e.touches[0].clientX - previousMousePosition.x; const deltaY = e.touches[0].clientY - previousMousePosition.y; const cosX = Math.cos(cubeRotationX); const directionModifier = cosX > 0 ? 1 : -1; cubeRotationY += deltaX * 0.01 * directionModifier; cubeRotationX += deltaY * 0.01; previousMousePosition = { x: e.touches[0].clientX, y: e.touches[0].clientY }; }); window.addEventListener('touchend', () => isDragging = false);
